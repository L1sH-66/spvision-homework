cmake_minimum_required(VERSION 3.8)
project(sp_nav_rviz_tools)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rviz_common REQUIRED)
find_package(rviz_rendering REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_library(maze_goal SHARED src/maze_goal.cpp)
ament_target_dependencies(maze_goal rviz_common rviz_rendering pluginlib rclcpp geometry_msgs)
target_link_libraries(maze_goal Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
install(TARGETS maze_goal LIBRARY DESTINATION lib)
ament_package()
